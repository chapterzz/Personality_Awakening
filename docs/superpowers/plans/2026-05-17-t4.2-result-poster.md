> **For agentic workers:** 用户批准 HTML 计划后，使用 superpowers:executing-plans 或 subagent-driven-development 按 Task 实施。实施前须 TDD。**禁止**在用户批准前修改 `apps/` 业务源码。

**Goal:** 在 `/report` 基于本地测评快照，用原生 Canvas 生成含 MBTI 类型、精灵、昵称与站点二维码的竖版 PNG，支持下载与可选 Web Share。

**Architecture:** 纯函数布局与 URL 构造 → `renderPosterToCanvas` 绘制 → `canvasToPngBlob` + `downloadBlob`；`PosterActions` 客户端组件挂到报告页；二维码库动态 import；无后端变更。

**Tech Stack:** Next.js 14 客户端组件、Canvas 2D API、`qrcode`、Vitest、可选 Playwright。

**Spec:** [2026-05-17-t4.2-result-poster-design.md](../specs/2026-05-17-t4.2-result-poster-design.md)

 * 解析海报二维码使用的站点 origin（T4.2）。

export function resolvePosterAppOrigin(fallbackOrigin: string): string {
  const env = typeof process !== 'undefined' ? process.env.NEXT_PUBLIC_APP_URL?.trim() : undefined;
  const base = env && env.length > 0 ? env : fallbackOrigin;
  return base.replace(/\/+$/, '');
}

```typescript
/**
 * 触发浏览器下载海报 PNG（T4.2）。
 */
export function downloadBlob(blob: Blob, filename: string): void {
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  a.rel = 'noopener';
  document.body.appendChild(a);
  a.click();
  a.remove();
  URL.revokeObjectURL(url);
}

export function buildPosterFilename(mbtiType: string, generatedAt: string): string {
  const day = generatedAt.slice(0, 10).replace(/-/g, '');
  return `ppa-poster-${mbtiType.toUpperCase()}-${day}.png`;
}
```

- [ ] **Step 2: 实现 sharePosterBlob（可选能力）**

```typescript
export async function sharePosterBlob(blob: Blob, mbtiType: string): Promise<boolean> {
  if (typeof navigator === 'undefined' || !navigator.share) return false;
  const file = new File([blob], buildPosterFilename(mbtiType, new Date().toISOString()), {
    type: 'image/png',
  });
  if (navigator.canShare && !navigator.canShare({ files: [file] })) return false;
  await navigator.share({
    files: [file],
    title: `我的 ${mbtiType.toUpperCase()} 探索卡`,
    text: '来自性格星球：觉醒计划',
  });
  return true;
}
```

```tsx
'use client';

/**
 * 报告页海报操作区：生成、预览、下载、可选分享（T4.2）。
 */
import { useCallback, useState } from 'react';
import { Button } from '@/components/ui/button';
import type { ReportSnapshot } from '@/lib/report-storage';
import { getStoredNickname } from '@/lib/auth-token';
import { resolvePosterAppOrigin } from '@/lib/poster-app-url';
import { renderPosterToCanvas, canvasToPngBlob } from '@/lib/poster-canvas';
import { downloadBlob, buildPosterFilename, sharePosterBlob } from '@/lib/poster-download';

const GUEST_NICKNAME = '星球探索者';

export function PosterActions({ snapshot }: { snapshot: ReportSnapshot }) {
  const [busy, setBusy] = useState(false);
  const [previewUrl, setPreviewUrl] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);

  const handleGenerate = useCallback(async () => {
    setBusy(true);
    setError(null);
    try {
      const nickname = getStoredNickname() ?? GUEST_NICKNAME;
      const origin = resolvePosterAppOrigin(window.location.origin);
      const canvas = await renderPosterToCanvas({
        mode: snapshot.mode,
        result: snapshot.result,
        generated_at: snapshot.generated_at,
        nickname,
        origin,
      });
      const blob = await canvasToPngBlob(canvas);
      if (previewUrl) URL.revokeObjectURL(previewUrl);
      setPreviewUrl(URL.createObjectURL(blob));
      (window as unknown as { __ppa_last_poster_blob?: Blob }).__ppa_last_poster_blob = blob;
    } catch {
      setError('海报生成失败，请重试');
    } finally {
      setBusy(false);
    }
  }, [snapshot, previewUrl]);

  const handleDownload = useCallback(() => {
    const blob = (window as unknown as { __ppa_last_poster_blob?: Blob }).__ppa_last_poster_blob;
    if (!blob) return;
    downloadBlob(blob, buildPosterFilename(snapshot.result.mbti_type, snapshot.generated_at));
  }, [snapshot]);

  // ... 渲染：生成按钮、下载按钮、分享按钮、preview img、朋友圈提示文案
}
```

（实施时移除 `window.__ppa_last_poster_blob` hack，改用 `useRef<Blob | null>`。）
