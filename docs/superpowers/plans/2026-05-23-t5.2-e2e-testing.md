- [ ] **Step 1: 写 auth-ui helper**

```typescript
/**
 * E2E 辅助：通过 /login UI 完成注册（可选演示 TestResult）。
 */
import { randomUUID } from 'crypto';

import { expect, type Page } from '@playwright/test';

export function uniqueNickname(prefix = 'e2e'): string {
  return `${prefix}-${randomUUID().slice(0, 8)}`;
}

export async function registerViaUi(
  page: Page,
  opts: {
    nickname: string;
    password: string;
    mbtiType?: string;
    withDemoResult?: boolean;
    redirect?: string;
  },
): Promise<void> {
  const redirect = opts.redirect ?? '/dashboard';
  await page.goto(`/login?redirect=${encodeURIComponent(redirect)}`);

  await page.getByRole('button', { name: '注册' }).click();
  await page.getByPlaceholder('2–32 个字符').fill(opts.nickname);
  await page.getByPlaceholder('至少 8 位').fill(opts.password);

  const withDemo = opts.withDemoResult ?? true;
  const demoCheckbox = page.getByRole('checkbox');
  if (withDemo) {
    await demoCheckbox.check();
    if (opts.mbtiType) {
      await page.locator('select').selectOption(opts.mbtiType);
    }
  } else {
    await demoCheckbox.uncheck();
  }

  await page.getByRole('button', { name: '注册' }).click();
  await expect(page).toHaveURL(new RegExp(`${redirect.replace(/\//g, '\\/')}$`), {
    timeout: 60_000,
  });
}
```

- [ ] **Step 1: 写 spec（复用 adaptive-standard-seed + mock scoring）**

```typescript
/**
 * Playwright：Golden Path — 标准测评完成 → 报告 → 导航看板（T5.2）。
 */
import { randomUUID } from 'crypto';

import { expect, test } from '@playwright/test';

import { seedGuestAdaptiveStandardComplete } from './helpers/adaptive-standard-seed';
import { expectDashboardLoaded } from './helpers/dashboard-assertions';

test('standard completed -> report -> dashboard navigation', async ({ page, request }) => {
  const sessionId = `e2e-golden-${randomUUID()}`;

  await seedGuestAdaptiveStandardComplete(request, sessionId);

  await page.addInitScript((sid: string) => {
    window.localStorage.setItem('ppa_guest_session_id', sid);
  }, sessionId);

  await page.route('**/scoring/mbti', async (route) => {
    await route.fulfill({
      status: 200,
      contentType: 'application/json',
      body: JSON.stringify({
        success: true,
        data: {
          mode: 'STANDARD',
          mbti_type: 'ENTJ',
          scores: {
            EI: { E: 8, I: 2, winner: 'E', delta: 6 },
            SN: { S: 3, N: 7, winner: 'N', delta: 4 },
            TF: { T: 7, F: 4, winner: 'T', delta: 3 },
            JP: { J: 6, P: 5, winner: 'J', delta: 1 },
          },
        },
        message: 'ok',
      }),
    });
  });

  await page.goto('/test/standard');
  await expect(page.getByRole('button', { name: '查看结果报告' })).toBeVisible({
    timeout: 60_000,
  });
  await page.getByRole('button', { name: '查看结果报告' }).click();
  await expect(page).toHaveURL(/\/report\?mode=STANDARD$/, { timeout: 60_000 });
  await expect(page.getByRole('heading', { level: 1, name: 'ENTJ' })).toBeVisible({
    timeout: 30_000,
  });

  await page.getByRole('link', { name: '看板' }).click();
  await expect(page).toHaveURL(/\/dashboard$/);
  await expectDashboardLoaded(page);
});
```
